function Read-NonEmptyString {
    param([Parameter(Mandatory=$true)][string]$Prompt, [string]$Default = "")
    while ($true) {
        $suffix = if ($Default) { " [$Default]" } else { "" }
        $val = Read-Host "$Prompt$suffix"
        if ([string]::IsNullOrWhiteSpace($val)) { if ($Default) { return $Default }; continue }
        return $val.Trim()
    }
}

function Invoke-ExpresswayApi {
    param($ExpresswayHost, $Credential, $Method, $Path)
    $uri = "https://$ExpresswayHost/api/v1/$($Path.Trim().TrimStart('/'))"
    $plainUser = $Credential.UserName
    $plainPass = $Credential.GetNetworkCredential().Password
    $base64 = [Convert]::ToBase64String([Text.Encoding]::UTF8.GetBytes("${plainUser}:${plainPass}"))
    $params = @{
        Uri         = $uri
        Method      = $Method
        Headers     = @{ "Accept" = "application/json"; "Authorization" = "Basic $base64" }
        ContentType = "application/json"
        TimeoutSec  = 20
    }
    Write-Host "Calling: $Method $uri" -ForegroundColor DarkGray
    return Invoke-RestMethod @params
}

function Save-Output {
    param($Object, $FilePath)
    if ($Object -is [string]) { $Object | Out-File $FilePath -Encoding UTF8 }
    else { $Object | ConvertTo-Json -Depth 50 | Out-File $FilePath -Encoding UTF8 }
}

# ---------------------------------------------------------------
# SETUP
# ---------------------------------------------------------------
Write-Host "`nExpressway X15 Connection Setup" -ForegroundColor Cyan
$ExpresswayHost = Read-NonEmptyString -Prompt "Expressway Host/IP" -Default "14.49.9.155"
$username       = Read-NonEmptyString -Prompt "Username" -Default "admin"
$cred           = New-Object System.Management.Automation.PSCredential ($username, (Read-Host "Password" -AsSecureString))

# Safe output directory: works when double-clicked OR run from a shell
$scriptRoot = if ($PSCommandPath) { Split-Path -Parent $PSCommandPath } else { $PWD.Path }
$outDir     = Join-Path $scriptRoot "output"
if (-not (Test-Path $outDir)) { New-Item -ItemType Directory $outDir | Out-Null }

# ---------------------------------------------------------------
# MAIN LOOP
# ---------------------------------------------------------------
while ($true) {
    Show-Menu | Write-Host
    switch (Read-Host "Select (1-7)") {
        "1" {
            $p = Read-NonEmptyString "Path (e.g. provisioning/common/sysinfo)"
            try { (Invoke-ExpresswayApi $ExpresswayHost $cred "GET" $p) | ConvertTo-Json -Depth 5 | Write-Host }
            catch {
                $msg = $_.Exception.Message
                if ($msg -match "404") {
                    Write-Host "`n  [404] No data returned from the Expressway." -ForegroundColor Red
                    Write-Host "  The server had no response for '$p'." -ForegroundColor Yellow
                    Write-Host "  This means the requested item does not exist on this device," -ForegroundColor Yellow
                    Write-Host "  or the filter value returned no matching records." -ForegroundColor Yellow
                    Write-Host "  Possible reasons:" -ForegroundColor Yellow
                    Write-Host "    - The filter value does not exist on this device (e.g. no acknowledged alarms means" -ForegroundColor Yellow
                    Write-Host "      querying alarms/status/acknowledged returns 404 because there is nothing there)" -ForegroundColor Yellow
                    Write-Host "    - The status filter value is not a valid option for this endpoint" -ForegroundColor Yellow
                    Write-Host "    - The path requires a valid name/ID in place of a {placeholder}" -ForegroundColor Yellow
                    Write-Host "    - The feature this endpoint relates to is not licensed or enabled on this device`n" -ForegroundColor Yellow
                } elseif ($msg -match "500") {
                    Write-Host "`n  [500] The Expressway returned an internal server error." -ForegroundColor Red
                    Write-Host "  This is a known Cisco API behaviour - a 500 is returned instead of" -ForegroundColor Yellow
                    Write-Host "  an empty result when a filter matches no records." -ForegroundColor Yellow
                    Write-Host "  Example: querying alarms/status/active when no active alarms exist." -ForegroundColor Yellow
                    Write-Host "  Try the base path without a filter to confirm data exists first.`n" -ForegroundColor Yellow
                } elseif ($msg -match "401") {
                    Write-Host "`n  [401] Authentication failed." -ForegroundColor Red
                    Write-Host "  The username or password was rejected by the Expressway.`n" -ForegroundColor Yellow
                } elseif ($msg -match "403") {
                    Write-Host "`n  [403] Access denied." -ForegroundColor Red
                    Write-Host "  Your account does not have permission to access this endpoint.`n" -ForegroundColor Yellow
                } else {
                    Write-Host "`n  [Error] $msg`n" -ForegroundColor Red
                }
            }
        }
        "2" {
            try {
                $r = Invoke-ExpresswayApi $ExpresswayHost $cred "GET" "provisioning/common/searchrule"
                Save-Output $r (Join-Path $outDir "searchrules_v15.json")
                Write-Host "Success: saved to output/searchrules_v15.json" -ForegroundColor Green
            } catch { Write-Host "Failed to pull search rules: $($_.Exception.Message)" -ForegroundColor Red }
        }
        "3" {
            $src = Join-Path $outDir "searchrules_v15.json"
            if (-not (Test-Path $src)) { Write-Host "Run option 2 first." -ForegroundColor Yellow; break }
            try {
                $json  = Get-Content $src -Raw | ConvertFrom-Json
                $lines = @("# SearchRules Merged", "searchrules:")
                $hasItems = $json | Get-Member -Name "items" -MemberType NoteProperty
                $rules = if ($hasItems) { $json.items } else { @($json) }
                foreach ($r in $rules) {
                    $hasName = $r | Get-Member -Name "name" -MemberType NoteProperty
                    $ruleName = if ($hasName) { $r.name } else { "" }
                    $lines += "  - name: ""$ruleName"""
                    foreach ($prop in $r.PSObject.Properties) {
                        if ($prop.Name -ne "name") {
                            $val = if ($null -eq $prop.Value) { "" }
                                   elseif ($prop.Value -is [string] -or $prop.Value -is [int] -or $prop.Value -is [bool]) { "$($prop.Value)" }
                                   else { $prop.Value | ConvertTo-Json -Compress -Depth 5 }
                            $lines += "    $($prop.Name): ""$val"""
                        }
                    }
                }
                $lines | Out-File (Join-Path $outDir "searchrules_merged.raml") -Encoding UTF8
                Write-Host "Created output/searchrules_merged.raml" -ForegroundColor Green
            } catch { Write-Host "Error building RAML file: $($_.Exception.Message)" -ForegroundColor Red }
        }
        "4" {
            Write-Host "`nPulling Call Policy Rules..." -ForegroundColor Cyan
            $paths = @("provisioning/common/callpolicy/rules", "provisioning/common/cpl/rule")
            $found = $false
            foreach ($p in $paths) {
                try {
                    $r = Invoke-ExpresswayApi $ExpresswayHost $cred "GET" $p
                    Save-Output $r (Join-Path $outDir "callpolicy_rules.json")
                    Write-Host "SUCCESS - Rules found at: $p" -ForegroundColor Green
                    $found = $true; break
                } catch { Write-Host "Path $p not available." -ForegroundColor Gray }
            }
            if (-not $found) { Write-Host "Call Policy rules are not exposed via the REST API on this platform." -ForegroundColor Red }
        }
        "5" {
            Write-Host "`nFetching Call Policy Table View..." -ForegroundColor Cyan
            $paths   = @("provisioning/common/callpolicy/rules", "provisioning/common/cpl/rule")
            $success = $false
            foreach ($p in $paths) {
                try {
                    $r     = Invoke-ExpresswayApi $ExpresswayHost $cred "GET" $p
                    $rules = if ($r.items) { $r.items } else { @($r) }
                    Write-Host "`nCall Policy Rules via $p`:" -ForegroundColor Green
                    $rules | Select-Object `
                        @{Name='Action';      Expression={$_.action}},
                        @{Name='Pattern';     Expression={$_.pattern}},
                        @{Name='Source';      Expression={$_.source}},
                        @{Name='Destination'; Expression={$_.destination}},
                        @{Name='State';       Expression={$_.state}} |
                        Format-Table -AutoSize
                    $success = $true; break
                } catch { continue }
            }
            if (-not $success) { Write-Host "Call Policy rules are not exposed via the REST API on this platform." -ForegroundColor Red }
        }
        "6" { Show-Help }
        "7" { return }
    }
}
